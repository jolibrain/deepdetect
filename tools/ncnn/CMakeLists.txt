cmake_minimum_required(VERSION 3.14)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
  project(deepdetect_ncnn_examples LANGUAGES CXX)
  find_package(ncnn 20261020 EXACT CONFIG REQUIRED)
endif()
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs)
foreach(example IN ITEMS yolox segformer)
  add_executable(ncnn_${example} ${example}.cc)
  target_compile_features(ncnn_${example} PRIVATE cxx_std_17)
  target_link_libraries(ncnn_${example} PRIVATE ncnn ${OpenCV_LIBS})
endforeach()
